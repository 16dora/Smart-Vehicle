cmake_minimum_required(VERSION 3.20)
project(smart_vehicle_cloud VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Drogon CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SODIUM REQUIRED IMPORTED_TARGET libsodium)
pkg_check_modules(SQLITE REQUIRED IMPORTED_TARGET sqlite3)
add_executable(smart_vehicle_cloud source/main.cpp source/cloud_service.cpp)
target_include_directories(smart_vehicle_cloud PRIVATE include)
target_link_libraries(smart_vehicle_cloud PRIVATE Drogon::Drogon PkgConfig::SQLITE PkgConfig::SODIUM)
target_compile_options(smart_vehicle_cloud PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS smart_vehicle_cloud RUNTIME DESTINATION bin)
install(DIRECTORY web/ DESTINATION share/smart-vehicle/web)
